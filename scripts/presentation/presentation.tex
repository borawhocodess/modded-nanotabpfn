\begin{frame}{Tabular foundation models, and why pretraining hurts}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{TabPFN-style TFMs} \cit{Hollmann '23/'25}
      \begin{itemize}
        \item pretrain a transformer on \emph{synthetic} datasets,
        \item then predict on a new task by \emph{in-context learning} --- no per-task training.

        \item slows iteration on architectures, priors, optimizers

\begin{frame}{The speedrun, in the spirit of modded-nanogpt}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Format} \cit{modded-nanogpt}
      \begin{itemize}
        \item one single-file training script
        \item contributors race to a fixed target in less wallclock
        \item the script logs its own source, config, versions, GPU, memory, timing
        \item every record reconstructs from its log
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Rules} (keep records comparable)
      \begin{enumerate}
        \item don't change the evaluation pipeline
        \item don't load pretrained weights
        \item must beat the prior record on the same hardware \& seed
      \end{enumerate}
      \medskip
      \mut{Everything else is fair game --- including the prior.}
    \end{column}
  \end{columns}
  \vfill
  \begin{block}{Timing protocol}
    \footnotesize Cumulative \textbf{training} wallclock only. Evaluation time is excluded (timer paused during validation). Prior generation is excluded (a pre-generated prior dump is allowed).
  \end{block}
\end{frame}

\section{Evaluation \& Method}

\begin{frame}{What ``hitting the target'' means}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Benchmark:} 38 TabArena classification tasks \cit{Erickson '25}.
      \medskip

      \textbf{Subsampling} (fast iteration):
      \begin{itemize}
        \item $>100$ features $\to$ keep 100 (random)
        \item $>1000$ rows $\to$ keep 1000 (stratified)
      \end{itemize}
      \medskip

      \textbf{Scoring:}
      \begin{itemize}
        \item 5-fold stratified CV, preprocessing fit on train only
        \item concat out-of-fold preds, score once
        \item ROC AUC (binary / one-vs-rest), mean over 38 tasks
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{The target}
        \[ \text{avg ROC AUC} \;\ge\; \hl{0.80685} \]
        = a scikit-learn \textbf{Random Forest} run through the \emph{identical} pipeline.
      \end{block}
      \medskip
      \begin{itemize}
        \item evaluation runs periodically during training
        \item training \textbf{stops} the instant the target is met
        \item everything seeded (seed 11): subsample, folds, splits
      \end{itemize}
      \medskip
      \mut{\footnotesize The target is hardcoded as the \texttt{jackpot} so every record races the same line.}
    \end{column}
  \end{columns}
\end{frame}

\section{Technique Deep-Dives}

\begin{frame}{Record 2 --- Muon optimizer \hfill \mut{54.41 min}}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Muon \cit{Jordan '24} orthogonalizes each 2D weight's gradient before the step:
      \[ W \mathrel{-}= \eta\,\mathrm{NS}(G), \quad \mathrm{NS}(G)\approx UV^\top \]
      with $G = U\Sigma V^\top$ --- it strips singular-value scaling, taking a unit-spectrum step.
      \medskip

      \begin{itemize}
        \item only 2D hidden matrices; AdamW keeps the rest
        \item orthogonalize via 5 \textbf{Newton--Schulz} steps (matmul-only, GPU-friendly)
        \item modded-nanogpt hyperparams; Muon LR $=0.1\times$ AdamW
      \end{itemize}
      \medskip
      \hl{Pure sample-efficiency win: 74.3 $\to$ 54.4 min.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.9]
        % raw gradient ellipse -> orthogonalized circle
        \draw[->,muted] (-2.4,0)--(2.4,0); \draw[->,muted] (0,-1.8)--(0,1.8);
        \draw[accent, very thick, rotate=25] (0,0) ellipse (2cm and 0.6cm);
        \node[accent, font=\scriptsize] at (1.7,1.3) {raw $G$};
        \draw[good, very thick] (0,0) circle (1.1cm);
        \node[good, font=\scriptsize] at (-1.0,1.3) {$\mathrm{NS}(G)$};
        \node[muted, font=\tiny] at (0,-2.4) {Newton--Schulz $\to$ equal singular values};
      \end{tikzpicture}
      \\[1mm]
      \mut{\scriptsize Conditioning the update: anisotropic gradient $\to$ isotropic step.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Record 3 --- SDPA, bf16, LR, width \hfill \mut{10.10 min}}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      A bundle of attention + precision + capacity changes:
      \begin{itemize}
        \item rewrite attention as \textbf{explicit-QKV SDPA} (fused kernel)
        \item \textbf{post- $\to$ pre-norm} blocks \cit{Xiong '20}
        \item LR $10^{-4}\!\to\!10^{-3}$
        \item embed $192\!\to\!256$, heads $6\!\to\!4$
        \item bf16 autocast + TF32 matmul
      \end{itemize}
      \medskip
      \hl{Biggest single cliff: 54.4 $\to$ 10.1 min.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Leave-one-out ablation} (median min):
      \scriptsize
      \begin{tabular}{@{}lrr@{}}
        \toprule
        Component & Muon$+$ & Rec$-$ \\
        \midrule
        LR $10^{-4}\!\to\!10^{-3}$ & 31.5 & \hl{+223\%} \\
        Explicit-QKV SDPA          & 39.0 & \hl{+121\%} \\
        Embed $(192,6)\!\to\!(256,4)$ & 45.8 & +53\% \\
        bf16 (train)               & 49.3 & +4\% \\
        bf16 (infer)               & 49.1 & $-$12\% \\
        TF32 matmul                & 45.1 & $-$6\% \\
        \bottomrule
      \end{tabular}
      \medskip\normalsize

      \mut{\footnotesize The \hl{LR increase} and \hl{SDPA rewrite} are the dominant drivers; precision tweaks are throughput trims.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Record 4 --- Batched Muon \& compile \hfill \mut{9.26 min}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Pure throughput, same training objective.
      \medskip
      \begin{itemize}
        \item \textbf{Batched Muon:} run the Newton--Schulz iteration \emph{once} across the stacked Q/K/V matrices instead of three separate calls
        \item \textbf{Compile:} \texttt{torch.compile} the encoder layer's forward
      \end{itemize}
      \medskip
      Dataset count unchanged (13{,}184) --- wallclock drops 9\%.
      \medskip
      \mut{\footnotesize This is why explicit-QKV (Record 3) mattered: it created the grouped matrix that makes batching possible.}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, node distance=2mm]
        \node (a) {3 separate NS calls};
        \node[draw, below=1mm of a, minimum width=8mm, fill=ink!10] (q1) {Q};

      \mut{\scriptsize One Newton--Schulz over a stacked tensor amortizes kernel-launch overhead.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Record 5 --- Residual decay \hfill \mut{7.57 min}}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Scale the input to block $i$ by an exponentially decaying factor:
      \[ h_i \;\leftarrow\; 0.95^{\,i}\cdot h_i \]
      Earlier-layer contributions are down-weighted in the final output.
      \medskip
      \begin{itemize}
        \item one scalar per layer, no params
        \item 9.26 $\to$ 7.57 min, 13.2k $\to$ 11.2k datasets
      \end{itemize}
      \medskip
      \mut{\footnotesize A re-run over 41 nodes later landed at 6.06 min median --- a reminder of cluster-noise variance.}
    \end{column}

\begin{frame}{Record 7 --- LAWA \& AdamW weight decay \hfill \mut{3.48 min}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Latest Weight Averaging} \cit{Kaddour '22; Sanyal '24}
      \begin{itemize}
        \item keep a FIFO of the last \textbf{10} checkpoints
        \item at \emph{evaluation only}, average them into a temporary model and score
        \item training continues on the un-averaged trajectory
      \end{itemize}
      \medskip
      Plus weight decay $0.01$ on the schedule-free AdamW group.
      \medskip
      \hl{3.88 $\to$ 3.48 min} --- cheap, eval-time-only smoothing.
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, node distance=1.5mm]
        \foreach \i/\x in {1/0,2/0.55,3/1.1,4/1.65,5/2.2} {

        }

\begin{frame}{Record 8 --- Repeated feature grouping \hfill \mut{2.15 min}}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      From TabICLv2 \cit{Qu '26}: embed each column \emph{jointly} with two others.
      \medskip

      Group $j$ = columns $(j,\, j{+}1,\, j{+}3)\bmod m$.
      \begin{itemize}
        \item shifts $(0,1,3)$ $\Rightarrow$ pairwise gaps $\{1,2,3\}$ all distinct
        \item so for $m\ge 7$ no pair co-occurs in two groups
        \item every column lives in 3 overlapping groups
      \end{itemize}
      \medskip
      Fixes representation collapse when features share distributions.
      \medskip
      \hl{3.48 $\to$ 2.15 min, 8.8k $\to$ 5.0k datasets.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize]
        \foreach \j/\lab in {0/c_0,1/c_1,2/c_2,3/c_3,4/c_4,5/c_5} {
          \node[draw, minimum size=6mm, fill=ink!8] (col\j) at (\j*0.8,0) {$\lab$};
        }
        % group starting at 0: cols 0,1,3
        \draw[accent, very thick, rounded corners]
          ($(col0.north west)+(-1mm,1mm)$) rectangle ($(col0.south east)+(1mm,-1mm)$);
        \node[accent] at (col0.north) [above=2mm] {group $j{=}0$};
        \draw[accent, -{Stealth[length=2mm]}] (col0.north) to[bend left=30] (col1.north);
        \draw[accent, -{Stealth[length=2mm]}] (col0.south) to[bend right=30] (col3.south);

      \end{tikzpicture}
      \\[2mm]
      \mut{\scriptsize Overlapping triples give each column three distinct joint contexts.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Record 9 --- Autoresearch HPO + two architecture changes \hfill \mut{0.92 min}}

      Plus two architecture changes:
      \begin{itemize}
        \item \textbf{Muon weight decay} ($0.1$)

      \medskip

\begin{frame}{Backup --- evaluation pipeline detail}
  \footnotesize
  \begin{itemize}
    \item \textbf{Subsampling:} $>100$ feats $\to$ 100 random; $>1000$ rows $\to$ 1000 stratified; fixed seed.
    \item \textbf{CV:} 5-fold \texttt{StratifiedKFold}, shuffled; labels integer-encoded per fold on train.
    \item \textbf{Preprocessing} (per fold, fit on train only):
      \begin{itemize}\footnotesize
        \item drop columns with $\le 1$ unique non-NaN value
        \item numeric: \texttt{to\_numeric} coercion + mean imputation
        \item categorical: ordinal encoding (unknown$\to$NaN) + most-frequent imputation
      \end{itemize}
    \item \textbf{Metric:} concat out-of-fold preds, \texttt{roc\_auc\_score} (binary / OvR), mean over 38 tasks.
    \item All randomness seeded with 11. Target $=0.8068462330697953$ (scikit-learn RandomForest through the same pipeline).
  \end{itemize}
\end{frame}
